  <script>
    function calcSummary(prefix) {
      const plan = parseFloat(document.getElementById('p_' + prefix).value) || 0;
      const act = parseFloat(document.getElementById('a_' + prefix).value) || 0;
      document.getElementById('g_' + prefix).value = plan - act;
    }

    function sumClass(className) {
      let sum = 0;
      document.querySelectorAll('.' + className).forEach(input => {
        sum += parseFloat(input.value) || 0;
      });
      return sum;
    }

    function calculateAll() {
      // Assembly line totals
      const assL1 = sumClass('ass-l1-act');
      const assL2 = sumClass('ass-l2-act');
      const assL3 = sumClass('ass-l3-act');
      document.getElementById('ass_l1_tot').value = assL1;
      document.getElementById('ass_l2_tot').value = assL2;
      document.getElementById('ass_l3_tot').value = assL3;
      document.getElementById('gt_ass').value = assL1 + assL2 + assL3;

      // Testing line totals
      const tstL1 = sumClass('tst-l1-act');
      const tstL2 = sumClass('tst-l2-act');
      const tstL3 = sumClass('tst-l3-act');
      document.getElementById('tst_l1_tot').value = tstL1;
      document.getElementById('tst_l2_tot').value = tstL2;
      document.getElementById('tst_l3_tot').value = tstL3;
      document.getElementById('gt_tst').value = tstL1 + tstL2 + tstL3;

      // F-off line totals
      const foL1 = sumClass('fo-l1-act');
      const foL2 = sumClass('fo-l2-act');
      const foL3 = sumClass('fo-l3-act');
      document.getElementById('fo_l1_tot').value = foL1;
      document.getElementById('fo_l2_tot').value = foL2;
      document.getElementById('fo_l3_tot').value = foL3;
      document.getElementById('gt_fo').value = foL1 + foL2 + foL3;

      // FEG line totals
      const fegL1 = sumClass('feg-l1-act');
      const fegL2 = sumClass('feg-l2-act');
      const fegL3 = sumClass('feg-l3-act');
      document.getElementById('feg_l1_tot').value = fegL1;
      document.getElementById('feg_l2_tot').value = fegL2;
      document.getElementById('feg_l3_tot').value = fegL3;
      document.getElementById('gt_feg').value = fegL1 + fegL2 + fegL3;
    }

    function exportCSV() {
      calculateAll();
      let csv = [];
      const rows = document.querySelectorAll("table tr");
      
      for (let i = 0; i < rows.length; i++) {
        let row = [], cols = rows[i].querySelectorAll("td, th");
        for (let j = 0; j < cols.length; j++) {
          let input = cols[j].querySelector("input");
          let val = input ? input.value : cols[j].innerText.trim();
          row.push('"' + val.replace(/"/g, '""') + '"');
        }
        csv.push(row.join(","));
      }

      const csvFile = new Blob([csv.join("\n")], { type: "text/csv" });
      const downloadLink = document.createElement("a");
      downloadLink.download = "production_record.csv";
      downloadLink.href = window.URL.createObjectURL(csvFile);
      downloadLink.style.display = "none";
      document.body.appendChild(downloadLink);
      downloadLink.click();
    }

    // Auto-calculate on input changes
    document.addEventListener('input', calculateAll);
  </script>
